{{- /* Helper: describes the series the page belongs to (the first `series` term) and returns a dict, or an empty dict when the page is not part of a series with at least two posts. Parts are ordered by date. Keys: term, parts, index (0-based), number (1-based), total, prev, next (prev/next are pages, or false). */ -}} {{- $info := dict -}} {{- with .GetTerms "series" -}} {{- $term := index . 0 -}} {{- $parts := $term.Pages.ByDate -}} {{- if gt (len $parts) 1 -}} {{- $index := 0 -}} {{- range $i, $part := $parts -}} {{- if $part.Eq $ -}}{{- $index = $i -}}{{- end -}} {{- end -}} {{- $prev := false -}} {{- if gt $index 0 -}}{{- $prev = index $parts (sub $index 1) -}}{{- end -}} {{- $next := false -}} {{- if lt (add $index 1) (len $parts) -}}{{- $next = index $parts (add $index 1) -}}{{- end -}} {{- $info = dict "term" $term "parts" $parts "index" $index "number" (add $index 1) "total" (len $parts) "prev" $prev "next" $next -}} {{- end -}} {{- end -}} {{- return $info -}}
