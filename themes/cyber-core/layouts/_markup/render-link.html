{{- /* Link render hook. External links (http/https to another host) get rel="noopener noreferrer", target="_blank" and class="external" (themes may add an indicator with a.external::after). Links produced by `ref` carry the site's own host and are therefore treated as internal. Relative destinations are resolved in this order: content page, page resource (a file in the bundle), global resource (assets/). One that matches nothing fails the build. Root-relative destinations are resolved to a page when possible and passed through otherwise (static files). Fragment-only links, mailto:, tel: and similar are passed through untouched. Destinations written with a shortcode, [text]({{< ref "posts/x" >}}), reach this hook as an internal placeholder (HAHAHUGOSHORTCODE…) that Hugo swaps for the real URL afterwards; it also validates the reference and fails the build when the page does not exist. They are therefore passed through here. */ -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- $siteHost := (urls.Parse site.BaseURL).Host -}} {{- $isWeb := in (slice "http" "https") $u.Scheme -}} {{- $external := and $isWeb (ne $u.Host $siteHost) -}} {{- $href := $dest -}} {{- $isShortcode := hasPrefix $dest "HAHAHUGOSHORTCODE" -}} {{- if and (not $isShortcode) (not $u.Scheme) (not $u.Host) $u.Path -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $rooted := hasPrefix $path "/" -}} {{- with or (.PageInner.GetPage $path) (.PageInner.GetPage (strings.TrimSuffix "/" $path)) -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery }}{{ $href = printf "%s?%s" $href . }}{{ end -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href . }}{{ end -}} {{- else with and (not $rooted) (or ($.PageInner.Resources.Get $path) (resources.Get $path)) -}} {{- $href = .RelPermalink -}} {{- else -}} {{- if not $rooted -}} {{- errorf "render-link: %q does not match a page or resource (%s). Use {{< relref \"section/page\" >}} or a path to the .md file." $dest $.Position -}} {{- end -}} {{- end -}} {{- end -}} {{- .Text -}} {{- /* chomp trailing newline */ -}}
